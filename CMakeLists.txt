if(ZHONGYUAN_BUILD_EXTENSION)
    set(ZHONGYUAN_EXTENSION_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/bin" CACHE PATH "Directory for the Godot extension (isolate diagnostic builds)")
    set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/native/build_profile.json" CACHE FILEPATH "" FORCE)
    set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "" FORCE)
    add_subdirectory(native/vendor/godot-cpp EXCLUDE_FROM_ALL)
    add_library(zhongyuan_gdextension SHARED native/godot/bridge.cpp native/godot/original_data.cpp native/godot/opening.cpp native/godot/register_types.cpp)
    target_link_libraries(zhongyuan_gdextension PRIVATE zhongyuan_core godot::cpp)
    target_include_directories(zhongyuan_gdextension PRIVATE native/godot)
    # Only the GDE_EXPORT entry point crosses the engine boundary. Keep bundled
    # C++ implementations private and resolve imports before the first call.
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_options(zhongyuan_gdextension PRIVATE "LINKER:--exclude-libs,ALL" "LINKER:-z,now")
    endif()
    set_target_properties(zhongyuan_gdextension PROPERTIES
        CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
        OUTPUT_NAME zhongyuan_core PREFIX ""
        LIBRARY_OUTPUT_DIRECTORY "${ZHONGYUAN_EXTENSION_OUTPUT_DIR}"
        RUNTIME_OUTPUT_DIRECTORY "${ZHONGYUAN_EXTENSION_OUTPUT_DIR}")
    foreach(config Debug Release RelWithDebInfo MinSizeRel)
        string(TOUPPER "${config}" upper_config)
        set_target_properties(zhongyuan_gdextension PROPERTIES
            LIBRARY_OUTPUT_DIRECTORY_${upper_config} "${ZHONGYUAN_EXTENSION_OUTPUT_DIR}"
            RUNTIME_OUTPUT_DIRECTORY_${upper_config} "${ZHONGYUAN_EXTENSION_OUTPUT_DIR}")
    endforeach()
endif()
